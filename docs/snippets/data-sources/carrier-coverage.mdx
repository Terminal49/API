export const CarrierCoverageStatus = ({ status, compact = false, availability = false, label }) => {
  const statuses = {
    supported: { label: availability ? "Available" : "Supported", symbol: "✓", className: "bg-green-50 text-green-700 dark:bg-green-500/10 dark:text-green-400" },
    limited: { label: availability ? "Limited" : "Partial", symbol: "◐", className: "bg-amber-50 text-amber-800 dark:bg-amber-500/10 dark:text-amber-300" },
    unsupported: { label: availability ? "Not available" : "Not supported", symbol: "−", className: "bg-gray-100 text-gray-600 dark:bg-white/10 dark:text-white/60" },
    unknown: { label: "Unknown", symbol: "?", className: "bg-gray-100 text-gray-600 dark:bg-white/10 dark:text-white/60" },
    signed_in_only: { label: "Requires carrier credentials", symbol: "⚿", className: "bg-blue-50 text-blue-700 dark:bg-blue-500/10 dark:text-blue-300" },
  }
  const value = statuses[status] || statuses.unknown
  return (
    <span title={label ? `${label}: ${value.label}` : value.label} className={`${label ? "shipping-feature-badge " : ""}inline-flex items-center justify-center gap-1.5 rounded-full px-2 py-1 text-xs font-medium ${value.className}`}>
      <span aria-hidden="true" className="shrink-0">{value.symbol}</span>
      {label ? <><span className="shipping-feature-label">{label}</span><span className="sr-only">, {value.label}</span></> : compact ? <span className="sr-only">{value.label}</span> : value.label}
    </span>
  )
}

export const CarrierCoverageDetails = ({ carrier, features, coverage, onClose, Status }) => {
  const groups = [...new Set(features.map((feature) => feature.group))]
  return (
    <section aria-label={`${carrier.name} feature coverage`} className="text-sm text-gray-900 dark:text-white">
      <div className="carrier-coverage-feature-groups grid">
        {groups.map((group) => (
          <div key={group} className="carrier-coverage-feature-group min-w-0">
            <h4 className="m-0 border-b border-gray-200 bg-gray-50 px-4 py-2 text-xs font-semibold dark:border-white/10 dark:bg-white/5">{group}</h4>
            <dl>
              {features.filter((feature) => feature.group === group).map((feature) => (
                <div key={feature.key} className="flex items-start justify-between gap-3 border-b border-gray-100 px-4 py-2.5 dark:border-white/10">
                  <dt className="min-w-0">
                    {feature.href ? <a href={feature.href} className="text-gray-900 underline decoration-gray-300 underline-offset-4 hover:text-green-700 dark:text-white dark:decoration-white/30 dark:hover:text-green-400">{feature.label}</a> : feature.label}
                    {feature.field ? <span className="mt-1 block break-all font-mono text-[11px] text-gray-500 dark:text-white/50">{feature.field}</span> : null}
                  </dt>
                  <dd className="shrink-0"><Status status={coverage?.features?.[feature.key]} /></dd>
                </div>
              ))}
            </dl>
          </div>
        ))}
      </div>
      <div className="flex justify-end border-t border-gray-200 px-4 py-2 dark:border-white/10">
        <button type="button" onClick={onClose} aria-label={`Close ${carrier.name} coverage details`} className="rounded px-2 py-1 text-xs text-gray-500 hover:bg-gray-100 focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600 dark:text-white/60 dark:hover:bg-white/10">Collapse details</button>
      </div>
    </section>
  )
}

export const ShippingCarrierCoverageDetails = ({ features, coverage, Status }) => {
  const routing = [
    { label: "Port of Lading (POL)", group: "Origin", estimated: "polEtd", actual: "polAtd", estimatedLabel: "ETD", actualLabel: "ATD" },
    { label: "Port of Discharge (POD)", group: "Destination port", estimated: "podEta", actual: "podAta", estimatedLabel: "ETA", actualLabel: "ATA" },
    { label: "Final Destination", group: "Inland destination", estimated: "destinationEta", actual: "destinationAta", estimatedLabel: "ETA", actualLabel: "ATA" },
  ]
  const sections = [
    { label: "Container Details", group: "Container attributes", open: false, hint: "Number · Seal · Equipment · Weight" },
    { label: "Core Transport Milestones", group: "Container milestones", open: true },
    { label: "Extended Transport Milestones", group: "Additional milestones", open: true },
    { label: "Import Milestones", keys: ["sslLfdChanged"], open: true },
    { label: "Other Events", group: "Other events", excluded: ["sslLfdChanged"], open: false },
  ]
  const FeatureBadges = ({ items }) => (
    <ul className="shipping-feature-badges">
      {items.map((feature) => (
        <li key={feature.key} data-feature-key={feature.key}>
          {feature.href ? <a href={feature.href} className="inline-flex max-w-full rounded-full hover:underline focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600"><Status status={coverage?.features?.[feature.key]} label={feature.label} availability /></a> : <Status status={coverage?.features?.[feature.key]} label={feature.label} availability />}
        </li>
      ))}
    </ul>
  )
  return (
    <div className="shipping-coverage-details space-y-3 p-3 text-sm text-gray-900 dark:text-white">
      <div className="flex flex-wrap items-center justify-between gap-3 px-1 py-1">
        <p className="font-medium">Data available through this carrier integration</p>
        <div className="flex flex-wrap gap-2" aria-label="Data availability legend">
          {["supported", "limited", "unsupported", "unknown", ...(Object.values(coverage?.features || {}).includes("signed_in_only") ? ["signed_in_only"] : [])].map((status) => <Status key={status} status={status} availability />)}
        </div>
      </div>
      <section aria-label="Routing" className="shipping-section">
        <h4 className="shipping-section-heading">Routing</h4>
        {routing.map((location) => (
          <details key={location.group} className="shipping-route">
            <summary className="shipping-route-summary">
              <span className="shipping-route-name">{location.label}</span>
              <span data-feature-key={location.estimated}><Status status={coverage?.features?.[location.estimated]} label={location.estimatedLabel} availability /></span>
              <span data-feature-key={location.actual}><Status status={coverage?.features?.[location.actual]} label={location.actualLabel} availability /></span>
              <span className="shipping-attribute-toggle text-xs text-gray-600 dark:text-white/70"><span className="shipping-when-closed">See all attributes</span><span className="shipping-when-open">Hide attributes</span><span className="shipping-chevron" aria-hidden="true">›</span></span>
            </summary>
            <div className="shipping-route-attributes px-3 pb-2">
              <h5 className="py-2 text-xs font-semibold text-gray-500 dark:text-white/60">{location.group}</h5>
              <FeatureBadges items={features.filter((feature) => feature.group === location.group && feature.key !== location.estimated && feature.key !== location.actual)} />
            </div>
          </details>
        ))}
      </section>
      {sections.map((section) => (
        <details key={section.label} className="shipping-section" open={section.open}>
          <summary className="shipping-section-heading"><span className="shipping-chevron" aria-hidden="true">›</span><span>{section.label}</span>{section.hint ? <span className="ml-auto text-xs font-normal text-gray-500 dark:text-white/60">{section.hint}</span> : null}</summary>
          <div className="p-3">
            <FeatureBadges items={features.filter((feature) => section.keys ? section.keys.includes(feature.key) : feature.group === section.group && !section.excluded?.includes(feature.key))} />
          </div>
        </details>
      ))}
    </div>
  )
}

export const ShippingCarrierCoverageRows = ({ rows, features, coverage, selectedScac, toggleRow, idPrefix, Status, Details }) => {
  const identifiers = [
    { key: "billOfLading", label: "MBOL", title: "Master bill of lading number" },
    { key: "bookingNumber", label: "Booking", title: "Booking number" },
    { key: "containerNumber", label: "Container", title: "Container number" },
  ]
  return (
    <div className="shipping-coverage-table rounded border border-gray-200 text-sm dark:border-white/15">
      <div className="shipping-carrier-columns shipping-table-heading bg-gray-50 text-xs font-medium text-gray-600 dark:bg-white/5 dark:text-white/70" aria-hidden="true"><span>Shipping line</span><span>Tracking number support</span><span>Details</span></div>
      {rows.map((carrier) => {
        const isSelected = selectedScac === carrier.scac
        const detailId = `${idPrefix}-details-${carrier.scac}`
        const buttonId = `${idPrefix}-toggle-${carrier.scac}`
        return (
          <article key={carrier.scac} data-carrier-scac={carrier.scac} className="shipping-carrier" aria-label={carrier.name}>
            <div className={`shipping-carrier-columns shipping-carrier-header ${isSelected ? "shipping-carrier-selected" : ""}`}>
              <div className="min-w-0"><span className="font-semibold text-gray-900 dark:text-white">{carrier.name}</span><div className="mt-1 break-words font-mono text-[11px] text-gray-500 dark:text-white/50">{carrier.scac}{carrier.alternativeScacs?.length ? ` · Also ${carrier.alternativeScacs.join(", ")}` : ""}</div></div>
              <div className="shipping-tracking-badges" aria-label="Tracking number support">
                {identifiers.map((identifier) => <span key={identifier.key} data-feature-key={identifier.key} title={identifier.title}><Status status={coverage[carrier.scac]?.features?.[identifier.key]} label={identifier.label} /></span>)}
              </div>
              <button id={buttonId} type="button" aria-expanded={isSelected} aria-controls={isSelected ? detailId : undefined} aria-label={`${isSelected ? "Hide" : "Show"} ${carrier.name} details`} onClick={() => toggleRow(carrier.scac)} className="shipping-details-toggle inline-flex items-center justify-end gap-2 rounded text-xs font-medium text-gray-700 focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600 dark:text-white/80"><span>{isSelected ? "Hide details" : "Show details"}</span><span aria-hidden="true">{isSelected ? "⌃" : "⌄"}</span></button>
            </div>
            {isSelected ? <section id={detailId} aria-labelledby={buttonId}><Details features={features} coverage={coverage[carrier.scac]} Status={Status} /></section> : null}
          </article>
        )
      })}
      {rows.length === 0 ? <p className="px-3 py-10 text-center text-gray-500 dark:text-white/60">No carriers match these filters. Change the search or required feature.</p> : null}
    </div>
  )
}

  const [requiredFeature, setRequiredFeature] = useState("all")
  const [includeLimited, setIncludeLimited] = useState(false)
  const [includeCredentials, setIncludeCredentials] = useState(false)
  const [selectedScac, setSelectedScac] = useState(null)
  const eligible = carriers.filter((carrier) => !access || carrier.access === access)
  const rows = eligible.filter((carrier) => {
    const matchesQuery = [carrier.name, carrier.legalName, carrier.scac, ...(carrier.alternativeScacs || [])].join(" ").toLowerCase().includes(query.trim().toLowerCase())
    const status = coverage[carrier.scac]?.features?.[requiredFeature]
    return matchesQuery && (requiredFeature === "all" || status === "supported" || (includeLimited && status === "limited") || (includeCredentials && status === "signed_in_only"))

  const summary = summaryKeys.map((key) => features.find((feature) => feature.key === key)).filter(Boolean)
  const idPrefix = `${carrierLabel}-${access || "all"}`.toLowerCase().replace(/[^a-z0-9-]/g, "-")
  const controlClass = "w-full rounded-lg border border-gray-200 bg-white px-3 py-2 text-sm text-gray-900 focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600 dark:border-white/10 dark:bg-white/5 dark:text-white"
  const changeFilter = (setter, value) => { setter(value); setSelectedScac(null) }
  const toggleRow = (scac) => setSelectedScac(selectedScac === scac ? null : scac)

  return (

        .shipping-carrier-selected { position: sticky; top: var(--carrier-coverage-sticky-top, 6rem); z-index: 10; border-bottom: 1px solid rgb(156 163 175 / 0.25); box-shadow: inset 3px 0 currentColor; }

        .shipping-section > summary:focus-visible, .shipping-route-summary:focus-visible { outline: 2px solid #16a34a; outline-offset: 2px; }
        .shipping-section[open] > summary, .shipping-section > h4 { border-bottom: 1px solid rgb(156 163 175 / 0.2); }

"Available fields may be absent for an individual container." : "Supported fields may be absent for an individual container."} <a href="/coverage/fields" className="underline underline-offset-2">See field availability</a>. Unknown means no feature status is available.</p>

We continuously improve our integrations, so coverage may have changed since this update. <a href="mailto:support@terminal49.com" className="underline underline-offset-2">Contact support</a> to confirm coverage for your carrier.</p> : null}
