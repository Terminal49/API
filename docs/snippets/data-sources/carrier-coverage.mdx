export const CarrierCoverageStatus = ({ status, compact = false, availability = false, label, description }) => {
  const statuses = {
    supported: { label: availability ? "Available" : "Supported", symbol: "✓", className: "bg-green-50 text-green-700 dark:bg-green-500/10 dark:text-green-400" },
    limited: { label: availability ? "Limited" : "Partial", symbol: "◐", className: "bg-amber-50 text-amber-800 dark:bg-amber-500/10 dark:text-amber-300" },
    unsupported: { label: availability ? "Not available" : "Not supported", symbol: "−", className: "bg-gray-100 text-gray-600 dark:bg-white/10 dark:text-white/60" },
    unknown: { label: "Unknown", symbol: "?", className: "bg-gray-100 text-gray-600 dark:bg-white/10 dark:text-white/60" },
    signed_in_only: { label: "Requires carrier credentials", symbol: "⚿", className: "bg-blue-50 text-blue-700 dark:bg-blue-500/10 dark:text-blue-300" },
  }
  const value = statuses[status] || statuses.unknown
  return (
    <span title={[label ? `${label}: ${value.label}` : value.label, description].filter(Boolean).join(". ")} className={`${label ? "shipping-feature-badge " : ""}inline-flex items-center justify-center gap-1.5 rounded-full px-2 py-1 text-xs font-medium ${value.className}`}>
      <span aria-hidden="true" className="shrink-0">{value.symbol}</span>
      {label ? <><span className="shipping-feature-label">{label}</span><span className="sr-only">, {value.label}{description ? `. ${description}` : ""}</span></> : compact ? <span className="sr-only">{value.label}</span> : value.label}
    </span>
  )
}

export const CarrierCoverageDetails = ({ features, coverage, Status }) => {
  const groups = [...new Set(features.map((feature) => feature.group))]
  return (
    <dl className="shipping-coverage-details text-sm text-gray-900 dark:text-white">
      {groups.map((group) => (
        <div key={group} className="shipping-feature-group">
          <dt>{group}</dt>
          <dd>
            <ul className="shipping-feature-badges">
              {features.filter((feature) => feature.group === group).map((feature) => {
                const description = [feature.field, ...(feature.requirements || []).map((requirement) => requirement === "rail_plan" ? "Requires the Rail Plan" : requirement)].filter(Boolean).join(". ")
                return (
                  <li key={feature.key} data-feature-key={feature.key}>
                    {feature.href ? <a href={feature.href} className="inline-flex max-w-full rounded-full hover:underline focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600"><Status status={coverage?.features?.[feature.key]} label={feature.label} description={description} availability /></a> : <Status status={coverage?.features?.[feature.key]} label={feature.label} description={description} availability />}
                  </li>
                )
              })}
            </ul>
          </dd>
        </div>
      ))}
    </dl>
  )
}

export const ShippingCarrierCoverageDetails = ({ features, coverage, Status }) => {
  const sections = [
    { label: "Port of Lading (POL)", group: "Origin", shortLabels: true },
    { label: "Port of Discharge (POD)", group: "Destination port", shortLabels: true },
    { label: "Final Destination", group: "Inland destination", shortLabels: true },
    { label: "Container Details", group: "Container attributes" },
    { label: "Core Transport Milestones", group: "Container milestones" },
    { label: "Extended Transport Milestones", group: "Additional milestones" },
    { label: "Other Events", group: "Other events" },
  ]
  const FeatureBadges = ({ items }) => (
    <ul className="shipping-feature-badges">
      {items.map((feature) => (
        <li key={feature.key} data-feature-key={feature.key}>
          {feature.href ? <a href={feature.href} className="inline-flex max-w-full rounded-full hover:underline focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600"><Status status={coverage?.features?.[feature.key]} label={feature.label} availability /></a> : <Status status={coverage?.features?.[feature.key]} label={feature.label} availability />}
        </li>
      ))}
    </ul>
  )
  return (
    <dl className="shipping-coverage-details text-sm text-gray-900 dark:text-white">
      {sections.map((section) => (
        <div key={section.group} className="shipping-feature-group">
          <dt>{section.label}</dt>
          <dd>
            <FeatureBadges items={features.filter((feature) => feature.group === section.group).map((feature) => section.shortLabels ? { ...feature, label: feature.label.replace(/^(POL|POD|Destination) /, "") } : feature)} />
          </dd>
        </div>
      ))}
    </dl>
  )
}

export const CarrierCoverageRows = ({ rows, features, coverage, selectedScac, toggleRow, idPrefix, summaryFeatures, summaryLabel, carrierLabel, availability, Status, Details }) => {
  return (
    <div className="shipping-coverage-table rounded border border-gray-200 text-sm dark:border-white/15">
      <div className="shipping-carrier-columns shipping-table-heading bg-gray-50 text-xs font-medium text-gray-600 dark:bg-white/5 dark:text-white/70" aria-hidden="true"><span>{carrierLabel}</span><span>SCACs</span><span>{summaryLabel}</span><span /></div>
      {rows.map((carrier) => {
        const isSelected = selectedScac === carrier.scac
        const detailId = `${idPrefix}-details-${carrier.scac}`
        const buttonId = `${idPrefix}-toggle-${carrier.scac}`
        return (
          <article key={carrier.scac} data-carrier-scac={carrier.scac} className={`shipping-carrier ${isSelected ? "shipping-carrier-expanded" : ""}`} aria-label={carrier.name}>
            <button id={buttonId} type="button" aria-expanded={isSelected} aria-controls={isSelected ? detailId : undefined} aria-label={`${isSelected ? "Hide" : "Show"} ${carrier.name} details`} aria-describedby={`${buttonId}-scacs ${buttonId}-tracking`} onClick={() => toggleRow(carrier.scac)} className={`shipping-carrier-columns shipping-carrier-header ${isSelected ? "shipping-carrier-selected" : ""}`}>
              <span className="shipping-carrier-name min-w-0 font-semibold text-gray-900 dark:text-white">{carrier.name}</span>
              <span id={`${buttonId}-scacs`} className="shipping-carrier-scacs break-words font-mono text-[11px] text-gray-500 dark:text-white/50">{[carrier.scac, ...(carrier.alternativeScacs || [])].join(", ")}</span>
              <span id={`${buttonId}-tracking`} className="shipping-tracking-badges">
                <span className="sr-only">{summaryLabel}: </span>
                {summaryFeatures.map((feature) => <span key={feature.key} data-feature-key={availability ? undefined : feature.key} data-summary-feature-key={availability ? feature.key : undefined} title={feature.title}><Status status={coverage[carrier.scac]?.features?.[feature.key]} label={feature.label} availability={availability} /></span>)}
              </span>
              <svg className="shipping-row-chevron" aria-hidden="true" width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="m9 6 6 6-6 6" /></svg>
            </button>
            {isSelected ? <section id={detailId} aria-labelledby={buttonId}><Details features={features} coverage={coverage[carrier.scac]} Status={Status} /></section> : null}
          </article>
        )
      })}
      {rows.length === 0 ? <p className="px-3 py-10 text-center text-gray-500 dark:text-white/60">No carriers match these filters. Change the search or required feature.</p> : null}
    </div>
  )
}

export const CarrierCoverageExplorer = ({ carriers, features: sourceFeatures, coverage = {}, summaryKeys, access, updatedOn, carrierLabel = "Carrier", layout = "rail" }) => {
  const features = layout === "rail" ? sourceFeatures.map((feature) => feature.key === "available" ? { ...feature, label: "Available for pickup" } : feature) : sourceFeatures
  const [query, setQuery] = useState("")
  const [requiredFeature, setRequiredFeature] = useState("all")
  const [includeLimited, setIncludeLimited] = useState(false)
  const [includeCredentials, setIncludeCredentials] = useState(false)
  const [selectedScac, setSelectedScac] = useState(null)
  const eligible = carriers.filter((carrier) => !access || carrier.access === access)
  const rows = eligible.filter((carrier) => {
    const matchesQuery = [carrier.name, carrier.legalName, carrier.scac, ...(carrier.alternativeScacs || [])].join(" ").toLowerCase().includes(query.trim().toLowerCase())
    const status = coverage[carrier.scac]?.features?.[requiredFeature]
    return matchesQuery && (requiredFeature === "all" || status === "supported" || (includeLimited && status === "limited") || (includeCredentials && status === "signed_in_only"))
  })
  const summary = layout === "shipping" ? [
    { key: "billOfLading", label: "MBOL", title: "Master bill of lading number" },
    { key: "bookingNumber", label: "Booking", title: "Booking number" },
    { key: "containerNumber", label: "Container", title: "Container number" },
  ] : summaryKeys.map((key) => features.find((feature) => feature.key === key)).filter(Boolean)
  const idPrefix = `${carrierLabel}-${access || "all"}`.toLowerCase().replace(/[^a-z0-9-]/g, "-")
  const controlClass = "w-full rounded-lg border border-gray-200 bg-white px-3 py-2 text-sm text-gray-900 focus-visible:outline focus-visible:outline-2 focus-visible:outline-green-600 dark:border-white/10 dark:bg-white/5 dark:text-white"
  const changeFilter = (setter, value) => { setter(value); setSelectedScac(null) }
  const toggleRow = (scac) => setSelectedScac(selectedScac === scac ? null : scac)
  const hasCredentials = eligible.some((carrier) => Object.values(coverage[carrier.scac]?.features || {}).includes("signed_in_only"))
  return (
    <div className="carrier-coverage not-prose min-w-0 space-y-4">
      <style>{`
        .carrier-coverage { container-type: inline-size; }
        .shipping-carrier + .shipping-carrier { border-top: 1px solid rgb(156 163 175 / 0.25); }
        .shipping-carrier-columns { display: grid; grid-template-columns: minmax(0, 1fr) auto; align-items: center; gap: 12px; padding: 14px 16px; }
        .shipping-table-heading { display: none; }
        .shipping-carrier-header { width: 100%; border: 0; text-align: left; font: inherit; color: inherit; cursor: pointer; background: rgb(var(--background-light, 255 255 255)); }
        .shipping-carrier-header:hover { background: #f9fafb; }
        .shipping-carrier-header:focus-visible { outline: 2px solid #16a34a; outline-offset: -2px; }
        .dark .shipping-carrier-header { background: rgb(var(--background-dark, 10 10 10)); }
        .dark .shipping-carrier-header:hover { background: #1f2937; }
        .shipping-carrier-expanded { background: rgb(var(--background-light, 255 255 255)); box-shadow: inset 3px 0 #737373; border-bottom: 2px solid #d4d4d4; }
        .shipping-carrier-expanded + .shipping-carrier { border-top: 0; }
        .shipping-carrier-header.shipping-carrier-selected { position: sticky; top: var(--carrier-coverage-sticky-top, 6rem); z-index: 10; background: #f0f0f0; border-bottom: 1px solid #d4d4d4; box-shadow: inset 3px 0 #737373; }
        .dark .shipping-carrier-expanded { background: rgb(var(--background-dark, 10 10 10)); border-bottom-color: #525252; box-shadow: inset 3px 0 #a3a3a3; }
        .dark .shipping-carrier-header.shipping-carrier-selected { background: #262626; border-bottom-color: #525252; box-shadow: inset 3px 0 #a3a3a3; }
        .shipping-carrier-name { grid-column: 1; grid-row: 1; }
        .shipping-carrier-scacs { grid-column: 1; grid-row: 2; }
        .shipping-tracking-badges { display: flex; grid-column: 1 / -1; grid-row: 3; flex-wrap: wrap; gap: 6px; font-size: 12px; }
        .shipping-row-chevron { grid-column: 2; grid-row: 1; justify-self: end; transition: transform 120ms; }
        .shipping-carrier-header[aria-expanded="true"] > .shipping-row-chevron { transform: rotate(90deg); }
        .shipping-coverage-details { margin: 0 12px 0 16px; }
        .shipping-feature-group { display: grid; grid-template-columns: minmax(0, 1fr); align-items: start; gap: 10px 16px; padding: 14px 16px; }
        .shipping-feature-group + .shipping-feature-group { border-top: 1px solid rgb(156 163 175 / 0.15); }
        .shipping-feature-group > dt { margin: 0; font-weight: 500; }
        .shipping-feature-group > dd { min-width: 0; margin: 0; }
        .shipping-feature-badges { display: flex; flex-wrap: wrap; gap: 8px; margin: 0; padding: 0; list-style: none; }
        .shipping-feature-badges > li { min-width: 0; max-width: 100%; }
        .shipping-feature-badge { max-width: 100%; padding: 6px 12px; text-align: left; }
        .shipping-feature-label { min-width: 0; overflow-wrap: anywhere; }
        @container (min-width: 640px) {
          .shipping-carrier-columns { grid-template-columns: minmax(0, 1.2fr) minmax(70px, 0.55fr) minmax(0, 1.8fr) 18px; }
          .shipping-table-heading { display: grid; border-bottom: 1px solid rgb(156 163 175 / 0.25); }
          .shipping-carrier-name, .shipping-carrier-scacs, .shipping-tracking-badges, .shipping-row-chevron { grid-column: auto; grid-row: auto; }
          .shipping-feature-group { grid-template-columns: minmax(0, 1fr) minmax(0, 2.6fr); }
        }
      `}</style>
      <div className="grid gap-3 sm:grid-cols-2">
        <label className="text-xs font-medium text-gray-600 dark:text-white/70">Search carriers
          <input type="search" value={query} onChange={(event) => changeFilter(setQuery, event.target.value)} placeholder="Carrier name or SCAC" className={`${controlClass} mt-1`} />
        </label>
        <label className="text-xs font-medium text-gray-600 dark:text-white/70">Required feature
          <select value={requiredFeature} onChange={(event) => changeFilter(setRequiredFeature, event.target.value)} className={`${controlClass} mt-1`}>
            <option value="all">All features</option>
            {features.map((feature) => <option key={feature.key} value={feature.key}>{feature.label}</option>)}
          </select>
        </label>
      </div>
      {requiredFeature !== "all" ? <div className="flex flex-wrap gap-4 text-sm text-gray-600 dark:text-white/70">
        <label className="flex items-center gap-2"><input type="checkbox" checked={includeLimited} onChange={(event) => changeFilter(setIncludeLimited, event.target.checked)} className="accent-green-600" />Include limited availability</label>
        {hasCredentials ? <label className="flex items-center gap-2"><input type="checkbox" checked={includeCredentials} onChange={(event) => changeFilter(setIncludeCredentials, event.target.checked)} className="accent-green-600" />Include features requiring carrier credentials</label> : null}
      </div> : null}
      <p className="text-xs text-gray-500 dark:text-white/60">Showing {rows.length} of {eligible.length} carriers. Select a carrier row for data availability.</p>
      <div className="flex flex-wrap items-center gap-2" aria-label="Data availability legend">
        <span className="mr-2 text-xs text-gray-600 dark:text-white/70">{layout === "shipping" ? "Data available through the carrier integration" : "Data available through the rail integration and Terminal49 estimates"}</span>
        {["supported", "limited", "unsupported", "unknown", ...(hasCredentials ? ["signed_in_only"] : [])].map((status) => <CarrierCoverageStatus key={status} status={status} availability />)}
      </div>
      <CarrierCoverageRows rows={rows} features={features} coverage={coverage} selectedScac={selectedScac} toggleRow={toggleRow} idPrefix={idPrefix} summaryFeatures={summary} summaryLabel={layout === "shipping" ? "Tracking number support" : "Data availability"} carrierLabel={carrierLabel} availability={layout !== "shipping"} Status={CarrierCoverageStatus} Details={layout === "shipping" ? ShippingCarrierCoverageDetails : CarrierCoverageDetails} />
      <p className="text-xs text-gray-500 dark:text-white/60">Available fields may be absent for an individual container. <a href="/coverage/fields" className="underline underline-offset-2">See field availability</a>. Unknown means no feature status is available.</p>
      {updatedOn ? <p className="text-xs leading-relaxed text-gray-500 dark:text-white/60">Data last updated {updatedOn}. We continuously improve our integrations, so coverage may have changed since this update. <a href="mailto:support@terminal49.com" className="underline underline-offset-2">Contact support</a> to confirm coverage for your carrier.</p> : null}
    </div>
  )
}
